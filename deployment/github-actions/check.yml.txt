name: Daily site health
on:
  push:
  pull_request:
  workflow_dispatch:
  schedule:
    - cron: '10 1 * * *'
permissions:
  contents: read
jobs:
  health:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '22'
      - run: npm run build
        env:
          SITE_URL: ${{ vars.SITE_URL }}
      - run: python3 scripts/release.py
      - run: npm test
      - run: npm run check
        env:
          CHECK_URL: ${{ vars.SITE_URL }}
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: daily-health-report
          path: reports/
